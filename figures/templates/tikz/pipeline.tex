\begin{tikzpicture}[node distance=5mm]
  \node (in)  {$(x, w)$};
  \node[box, right=of in] (s1) {Witness\\[-1pt]trace};
  \node[box, right=of s1] (s2) {LDE};
  \node[ours, right=of s2] (s3) {Commit\\[-1pt]{\footnotesize (streamed)}};

  \foreach \a/\b in {in/s1, s1/s2, s2/s3, s3/s4, s4/s5, s5/out} \draw[flow] (\a) -- (\b);
  % data held in memory, all on one baseline above the boxes
  \coordinate (lv) at ($(s3.north)+(0,3mm)$);
  \foreach \n/\l in {s1/{nw}, s2/{nw/\rho}, s3/{\sqrt{n}\,w}, s4/{n/\rho}, s5/{n/\rho}}

  \node[op, anchor=base east, text=crmuted] at ($(in.west |- lv)+(-1mm,0)$) {memory};
  % claimed saving
  \draw[brace, decoration={mirror}] ($(s2.south west)+(0,-1mm)$) -- node[op, below=5pt] {peak memory: $O(\sqrt{n}\,w)$ instead of $O(nw/\rho)$}
